    <script>
        // Set dynamic year in footer
        document.getElementById('year').textContent = new Date().getFullYear();

        // Set default date picker to today
        const today = new Date().toISOString().split('T')[0];
        document.getElementById('resDate').setAttribute('min', today);
        document.getElementById('resDate').value = today;

        // Mobile Menu Drawer Toggle
        function toggleMobileMenu() {
            const menu = document.getElementById('mobileMenu');
            const icon = document.getElementById('menuIcon');
            menu.classList.toggle('hidden');
            if (menu.classList.contains('hidden')) {
                icon.className = "fa-solid fa-bars text-xl";
            } else {
                icon.className = "fa-solid fa-xmark text-xl";
            }
        }

        // Switchable Set Menu Tabs
        function switchMenu(type) {
            const tabSriLankan = document.getElementById('tabSriLankan');
            const tabWestern = document.getElementById('tabWestern');
            const menuSriLankan = document.getElementById('menuSriLankan');
            const menuWestern = document.getElementById('menuWestern');

            if (type === 'srilankan') {
                tabSriLankan.className = "px-6 py-3 rounded-full text-xs sm:text-sm font-semibold tracking-wider transition-all duration-300 bg-gold text-espresso-dark shadow-md";
                tabWestern.className = "px-6 py-3 rounded-full text-xs sm:text-sm font-semibold tracking-wider transition-all duration-300 text-cream/70 hover:text-gold";
                menuSriLankan.classList.remove('hidden');
                menuWestern.classList.add('hidden');
            } else {
                tabWestern.className = "px-6 py-3 rounded-full text-xs sm:text-sm font-semibold tracking-wider transition-all duration-300 bg-gold text-espresso-dark shadow-md";
                tabSriLankan.className = "px-6 py-3 rounded-full text-xs sm:text-sm font-semibold tracking-wider transition-all duration-300 text-cream/70 hover:text-gold";
                menuWestern.classList.remove('hidden');
                menuSriLankan.classList.add('hidden');
            }
        }

        // Lightbox Modal Functions
        function openLightbox(imgSrc, caption) {
            const modal = document.getElementById('lightboxModal');
            const img = document.getElementById('lightboxImg');
            const cap = document.getElementById('lightboxCaption');
            img.src = imgSrc;
            cap.textContent = caption;
            modal.classList.remove('hidden');
            modal.classList.add('flex');
        }

        function closeLightbox() {
            const modal = document.getElementById('lightboxModal');
            modal.classList.add('hidden');
            modal.classList.remove('flex');
        }

        // WhatsApp Booking Modal Functions
        function openBookingModal(presetMenu) {
            const modal = document.getElementById('bookingModal');
            if (presetMenu) {
                document.getElementById('resMenu').value = presetMenu;
            }
            modal.classList.remove('hidden');
            modal.classList.add('flex');
        }

        function closeBookingModal() {
            const modal = document.getElementById('bookingModal');
            modal.classList.add('hidden');
            modal.classList.remove('flex');
        }

        // Handle Reservation WhatsApp Formatting
        function handleReservation(e) {
            e.preventDefault();
            const name = document.getElementById('resName').value;
            const date = document.getElementById('resDate').value;
            const guests = document.getElementById('resGuests').value;
            const menu = document.getElementById('resMenu').value;
            const notes = document.getElementById('resNotes').value || 'None';

            const message = `*Table Reservation Request - The Rustique Restaurant*%0A%0A` +
                            `*Name:* ${encodeURIComponent(name)}%0A` +
                            `*Date:* ${encodeURIComponent(date)}%0A` +
                            `*Party Size:* ${encodeURIComponent(guests)}%0A` +
                            `*Set Menu Choice:* ${encodeURIComponent(menu)}%0A` +
                            `*Special Requests:* ${encodeURIComponent(notes)}%0A%0A` +
                            `_Sent via Rustique Website_`;

            // Redirects to WhatsApp text trigger
            window.open(`https://wa.me/?text=${message}`, '_blank');
            closeBookingModal();
        }
    </script>
